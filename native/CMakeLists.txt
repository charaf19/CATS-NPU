cmake_minimum_required(VERSION 3.16)
project(cats_native_topk LANGUAGES CXX)
add_library(cats_topk SHARED topk.cpp)
target_compile_features(cats_topk PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(cats_topk PRIVATE /O2 /W4)
else()
  target_compile_options(cats_topk PRIVATE -O3 -Wall -Wextra -Wpedantic)
endif()
set_target_properties(cats_topk PROPERTIES OUTPUT_NAME cats_topk PREFIX "")
